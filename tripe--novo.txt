<script>
/* ════════════════════════════════════

const questions = [
  {
    text: "Qual é a sua idade?",
    options: ["Entre 18 e 25 anos","Entre 26 e 35 anos","Entre 36 e 50 anos","Entre 51 e 65 anos"]
  },
  {
    text: "Qual é o tamanho aproximado do seu membro em ereção?",
    options: ["Menos de 12 cm","De 12 a 14 cm","De 14 a 16 cm","Mais de 16 cm"]
  },
  {
    text: "Você costuma acordar com ereção matinal?",
    options: ["Quase nunca","Raramente","Às vezes","Quase sempre"]
  },
  {
    text: "Como é a sua alimentação diária?",
    options: ["Só comida ruim","50% comida ruim, 50% saudável","Tento comer bem na maioria das vezes","Dieta equilibrada e saudável"]
  },
  {
    text: "Quantas horas você dorme por noite?",
    options: ["Menos de 5 horas","De 5 a 6 horas","De 6 a 7 horas","Mais de 8 horas"]
  },
  {
    text: "Você pratica atividade física?",
    options: ["Nunca","1 vez por semana ou menos","De 2 a 3 vezes por semana","4 vezes ou mais por semana"]
  },
  {
    text: "Você já tentou algum método natural para aumentar o membro?",
    options: ["Nunca ouvi falar","Já ouvi falar, mas nunca tentei","Já tentei, mas sem consistência","Sim, com disciplina"]
  }
];

const progressSteps = [14, 29, 43, 57, 71, 86, 100];

let currentQ = 0;
let answers = new Array(questions.length).fill(null);

/* ════════════════════════════════════
   RENDER QUESTION
════════════════════════════════════ */
function renderQuestion(idx) {
  const q = questions[idx];
  const isFirst = idx === 0;
  const isLast  = idx === questions.length - 1;
  const pct     = progressSteps[idx];

  // Header
  document.getElementById('progress-label').textContent = `Pergunta ${idx+1} de 7`;
  document.getElementById('progress-pct').textContent   = `${pct}% completo`;
  document.getElementById('progress-bar').style.width   = pct + '%';

  // Body
  const body = document.getElementById('quiz-body');
  body.style.animation = 'none';
  void body.offsetWidth;
  body.style.animation = '';

  let optionsHTML = q.options.map((opt, i) => `
    <button class="option-btn${answers[idx] === i ? ' selected' : ''}"
            onclick="selectOption(${i})"
            data-idx="${i}">
      <div class="radio-circle"></div>
      <span class="option-label">${opt}</span>
    </button>
  `).join('');

  body.innerHTML = `
    <div class="question-text">${q.text}</div>
    <div class="options-list">${optionsHTML}</div>
  `;

  // Footer
  const footer = document.getElementById('quiz-footer');
  const hasAnswer = answers[idx] !== null;

  if (isFirst) {
    footer.innerHTML = `
      <button class="btn-next-solo ${hasAnswer ? 'enabled' : ''}" id="btn-next" onclick="nextQuestion()">
        Continuar &nbsp;→
      </button>
    `;
  } else if (isLast) {
    footer.innerHTML = `
      <button class="btn-back" onclick="prevQuestion()">← &nbsp;Voltar</button>
      <button class="btn-next ${hasAnswer ? 'enabled' : ''}" id="btn-next" onclick="nextQuestion()">
        ✨ Ver meu potencial
      </button>
    `;
  } else {
    footer.innerHTML = `
      <button class="btn-back" onclick="prevQuestion()">← &nbsp;Voltar</button>
      <button class="btn-next ${hasAnswer ? 'enabled' : ''}" id="btn-next" onclick="nextQuestion()">
        Continuar &nbsp;→
      </button>
    `;
  }
}

function selectOption(i) {
  answers[currentQ] = i;
  // Update UI
  document.querySelectorAll('.option-btn').forEach((btn, idx) => {
    btn.classList.toggle('selected', idx === i);
    btn.querySelector('.radio-circle').style.borderColor = idx === i ? 'var(--cyan)' : '';
    btn.querySelector('.radio-circle').style.background  = idx === i ? 'var(--cyan)' : '';
  });
  // Enable next btn
  const btn = document.getElementById('btn-next');
  if (btn) btn.classList.add('enabled');

  // Auto-advance after short delay
  setTimeout(() => nextQuestion(), 420);
}

function nextQuestion() {
  if (answers[currentQ] === null) return;
  if (currentQ < questions.length - 1) {
    currentQ++;
    renderQuestion(currentQ);
  } else {
    showLoading();
  }
}

function prevQuestion() {
  if (currentQ > 0) {
    currentQ--;
    renderQuestion(currentQ);
  }
}

   VIDEO PLAYER
════════════════════════════════════ */
const VIDEO_DURATION = 102; // 1:42 em segundos
let videoUnlocked = false;
let countdownInterval = null;

function initVideo() {
  const video = document.getElementById('main-video');

  video.addEventListener('timeupdate', () => {
    const pct = (video.currentTime / (video.duration || VIDEO_DURATION)) * 100;
    document.getElementById('seek-fill').style.width = pct + '%';
    document.getElementById('time-display').textContent =
      formatTime(video.currentTime) + ' / ' + formatTime(video.duration || VIDEO_DURATION);

    // Unlock when video ends (or 95% watched)
    if (!videoUnlocked && pct >= 98) {
      unlockCTA();
    }
  });

  video.addEventListener('play', () => {
    document.getElementById('icon-play').style.display  = 'none';
    document.getElementById('icon-pause').style.display = 'block';
  });

  video.addEventListener('pause', () => {
    document.getElementById('icon-play').style.display  = 'block';
    document.getElementById('icon-pause').style.display = 'none';
  });

  video.addEventListener('ended', () => {
    document.getElementById('icon-play').style.display  = 'block';
    document.getElementById('icon-pause').style.display = 'none';
    unlockCTA();
  });

  // Start playing
  video.play().catch(() => {});
}

function togglePlay() {
  const video = document.getElementById('main-video');
  if (video.paused) video.play();
  else video.pause();
}

function toggleMute() {
  const video = document.getElementById('main-video');
  video.muted = !video.muted;
  document.getElementById('icon-vol').style.display  = video.muted ? 'none'  : 'block';
  document.getElementById('icon-mute').style.display = video.muted ? 'block' : 'none';
}

function seekVideo(e) {
  const video = document.getElementById('main-video');
  const bar   = document.getElementById('seek-bar');
  const rect  = bar.getBoundingClientRect();
  const ratio = Math.max(0, Math.min(1, (e.clientX - rect.left) / rect.width));
  video.currentTime = ratio * (video.duration || VIDEO_DURATION);
}

function formatTime(s) {
  if (isNaN(s)) return '0:00';
  const m = Math.floor(s / 60);
  const sec = Math.floor(s % 60);
  return m + ':' + String(sec).padStart(2,'0');
}

function unlockCTA() {
  if (videoUnlocked) return;
  videoUnlocked = true;
  const btn = document.getElementById('cta-btn');
  btn.classList.add('unlocked');
  btn.innerHTML = `
    <svg width="18" height="18" viewBox="0 0 24 24" fill="currentColor">
      <path d="M12 1L3 5v6c0 5.55 3.84 10.74 9 12 5.16-1.26 9-6.45 9-12V5l-9-4zm-2 16l-4-4 1.41-1.41L10 14.17l6.59-6.59L18 9l-8 8z"/>
    </svg>
    Quero meu acesso agora →
  `;
  btn.style.cursor = 'pointer';
}

function goToCheckout() {
  if (!videoUnlocked) return;
  window.location.href = 'https://paylume.fans/c/tripe-luc-org';
}
